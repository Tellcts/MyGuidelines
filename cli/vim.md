## 一、模式切换（Vim 的一切基础）

Vim 有三大核心模式：**普通模式**（默认）、**插入模式**（输文字）、**可视模式**（选文本）。

```vim
" 从普通模式进入插入模式
i        " 在光标前插入（最常用）
a        " 在光标后插入
I        " 在行首插入
A        " 在行尾插入
o        " 在下一行插入新行
O        " 在上一行插入新行

" 回到普通模式（最最最重要的按键！）
Esc      " 任何时候按 Esc 都能回到普通模式
<C-[>    " Ctrl+[ 等价于 Esc（手不用离开主键盘区）

" 进入可视模式（用于选中文本）
v        " 字符可视模式（选中一个或多个字符）
V        " 行可视模式（选中整行）
<C-v>    " 块可视模式（选中矩形区域）
```

> **记忆口诀**：`i` = insert（插入），`a` = append（追加），`o` = open（打开新行）

## 二、移动光标（每天用八百次）

### 1. 基础移动

```vim
h        " ← 左移一个字符
j        " ↓ 下移一行
k        " ↑ 上移一行
l        " → 右移一个字符
```

> **为什么不是方向键？** `hjkl` 让你不用把手挪到方向键区，保持右手在主键盘上。

### 2. 词级移动（比 h/l 有效率得多）

```vim
w        " 跳到下一个单词的开头（word）
b        " 跳回上一个单词的开头（back）
e        " 跳到当前/下一个单词的结尾（end）
ge       " 跳回上一个单词的结尾

W        " 大写 = 以空格分隔的"大词"跳跃（跳过标点）
B        " 同上，反向
E        " 同上，词尾
```

### 3. 行内移动

```vim
0        " 跳到行首（第 0 列）
^        " 跳到行首第一个非空字符
$        " 跳到行尾
g_       " 跳到行尾最后一个非空字符
f{字符}   " 跳到本行中下一个指定字符的位置（如 fa 跳到下一个 a）
F{字符}   " 同上，反向搜索
t{字符}   " 跳到本行中下一个指定字符的前一个字符（till）
T{字符}   " 同上，反向
;        " 重复上一次 f/F/t/T 操作
,        " 反向重复
```

### 4. 页面级移动

```vim
gg       " 跳到文件第一行
G        " 跳到文件最后一行
{行号}G  " 跳到指定行（如 42G 跳到第 42 行）
{行号}gg " 同上（42gg）

<C-f>    " Page Down 翻下一页（Forward）
<C-b>    " Page Up 翻上一页（Backward）
<C-d>    " 向下翻半页（Down）
<C-u>    " 向上翻半页（Up）

zz       " 将当前行置于屏幕中央（最实用）
zt       " 将当前行置于屏幕顶部（top）
zb       " 将当前行置于屏幕底部（bottom）

H        " 跳到屏幕顶端（Home/High）
M        " 跳到屏幕中间（Middle）
L        " 跳到屏幕底部（Low）
```

### 5. 段落与代码块

```vim
{        " 跳到上一段（空行分隔）
}        " 跳到下一段
%        " 跳转到匹配的括号 ({[]})（检查括号配对神器）
```

## 三、编辑操作（增删改查）

### 6. 删除

```vim
x        " 删除光标下的一个字符（像剪刀）
dd       " 删除当前整行（最常用的删除）
D        " 从光标删到行尾（等价于 d$）
d0       " 从光标删到行首
dw       " 删除一个单词（delete word）
d$       " 从光标删到行尾
dG       " 从当前行删到文件末尾
dgg      " 从当前行删到文件开头

" 数字 + 操作（Vim 最强大的组合技）
3dd      " 删除 3 行（从当前行开始）
5dw      " 删除 5 个单词
```

### 7. 复制（Vim 里叫"拉取"，yank）

```vim
yy       " 复制当前行（最常用）
Y        " 复制当前行（等价于 yy）
yw       " 复制一个单词（yank word）
y$       " 从光标复制到行尾
y0       " 从光标复制到行首
yG       " 从当前行复制到文件末尾
3yy      " 复制 3 行
```

### 8. 粘贴

```vim
p        " 在光标后粘贴（paste，小写 = 后/下）
P        " 在光标前粘贴（大写 = 前/上）
```

### 9. 撤销与重做

```vim
u        " 撤销上一次操作（undo，救命键！）
<C-r>    " 重做（redo，撤销了又后悔？按这个）
U        " 撤销当前行的所有修改（超大号后悔药）
```

### 10. 替换

```vim
r{字符}   " 替换光标下的一个字符（不用进入插入模式）
R        " 进入替换模式（连续替换，像在 Word 里按 Insert）
~        " 切换光标下字符的大小写
```

## 四、进入插入模式的各种姿势

```vim
i        " 在光标前插入
a        " 在光标后插入
I        " 在行首插入
A        " 在行尾插入
o        " 在当前行下方新建一行插入
O        " 在当前行上方新建一行插入
s        " 删除光标下字符并进入插入模式
S        " 删除当前行并进入插入模式
cc       " 删除当前行并进入插入模式（等价于 S）
C        " 从光标删到行尾并进入插入模式
cw       " 删除一个单词并进入插入模式
ciw      " 删除光标所在的整个单词并进入插入模式（不管光标在哪）
```

> **`ciw` 是宝藏命令**：光标在单词中间想改整个单词？`ciw` 一键搞定。

## 五、选择与操作（可视模式）

```vim
" 进入可视模式
v        " 字符选择
V        " 行选择
<C-v>    " 块选择（矩形区域——批量编辑神器）

" 选中后可以做的操作
d        " 删除选中内容
y        " 复制选中内容
c        " 删除选中内容并进入插入模式
>        " 向右缩进（缩进代码时用）
<        " 向左缩进
~        " 切换大小写
gu       " 变小写
gU       " 变大写

" 常用技巧
" 1. 按 V 选中多行，按 = 自动缩进（格式化代码）
" 2. <C-v> 选中多行开头，按 I 可以批量插入
```

### 块选择实战：批量添加注释

```vim
" 给多行加注释（比如在 Python 每行前加 #）
1. 光标移到想开始的位置
2. 按 <C-v> 进入块选择模式
3. 按 jjjj 向下选若干行
4. 按 I（大写 i），输入 #，按 Esc
   → 所有选中的行都加上了 #
```

## 六、搜索与替换

### 11. 搜索

```vim
/{关键词}  " 向下搜索（如 /hello 按下回车就开始搜）
?{关键词}  " 向上搜索
n        " 跳到下一个匹配结果
N        " 跳到上一个匹配结果
*        " 搜索光标下的单词（向下，# 是向上）
#        " 搜索光标下的单词（向上）
```

### 12. 搜索高亮控制

```vim
:noh     " 取消搜索高亮（noh = no highlight）
```

### 13. 查找与替换

```vim
" 基本替换（当前行第一个匹配项）
:s/旧/新

" 当前行所有匹配项
:s/旧/新/g

" 全文替换（要确认）
:%s/旧/新/gc

" 全文替换（不确认，直接换）
:%s/旧/新/g

" 全文中替换（包括被替换的词本身）
:%s/旧/新/ge

" 带范围的替换（第 10 到 20 行）
:10,20s/旧/新/g

" 替换时确认（c = confirm）
:%s/旧/新/gc
" 会逐一询问：y(替换) n(跳过) a(全部) q(退出)

" 使用正则
:%s/\<word\>/term/g   " 只替换完整的 word（\< \> 表示单词边界）
```

## 七、缓冲区、多窗口与标签页

### 14. 分屏操作

```vim
" 水平分屏（上下）
:sp {文件名}    " split
<C-w>s          " 快捷键分屏

" 垂直分屏（左右）
:vsp {文件名}   " vertical split
<C-w>v          " 快捷键分屏

" 分屏间导航
<C-w>h          " 跳到左边窗口
<C-w>j          " 跳到下面窗口
<C-w>k          " 跳到上面窗口
<C-w>l          " 跳到右边窗口
<C-w>w          " 循环切换窗口

" 调整窗口大小
<C-w>=          " 使所有窗口等宽等高
<C-w>_          " 最大化当前窗口高度
<C-w>|          " 最大化当前窗口宽度
<C-w>+          " 增加当前窗口高度
<C-w>-          " 减少当前窗口高度

" 关闭分屏
:q             " 关闭当前窗口
:only          " 只保留当前窗口（关闭其他所有）
```

### 15. 标签页

```vim
:tabnew {文件}   " 新建标签页
:tabn           " 下一个标签页（tabnext）
:tabp           " 上一个标签页（tabprevious）
:tabfir         " 跳转第一个标签页
:tablast        " 跳转最后一个标签页
gt              " 下一个标签页（图形界面快捷键）
gT              " 上一个标签页
{数字}gt        " 跳转到第 N 个标签页
:tabclose       " 关闭当前标签页
:tabs           " 列出所有标签页
```

## 八、文件操作

```vim
:e {文件}      " 打开文件（edit）
:w            " 保存文件（write，最常用）
:q            " 退出（quit）
:wq           " 保存并退出（最常用）
:x            " 保存并退出（只有在做了修改时才写文件）
:q!           " 强制退出（不保存）
:w!           " 强制保存（只读文件等）
:wq!          " 强制保存并退出
ZZ            " 保存并退出（不用按 : —— 普通模式下直接按大写 Z 两次）
ZQ            " 不保存退出

:e!           " 重新加载文件（放弃当前所有改动）
:saveas {路径} " 另存为
:f {文件名}    " 重命名当前文件

" 文件浏览器
:E            " 打开内置文件浏览器（Explore）
:Sexploit     " 在分屏中打开文件浏览器
`

" 读入其他文件内容
:r {文件名}    " 将文件内容插入当前光标处
:r !命令       " 将命令输出插入当前光标处（如 :r !date 插入当前时间）
```

## 九、常用编辑技巧

### 16. 重复与宏

```vim
.        " 重复上一次修改（超级无敌实用！）
         " 比如你刚刚做了 ciwhello，按 . 就再改一个词为 hello

" 录制宏（自动化操作）
q{寄存器}  " 开始录制宏到指定寄存器（如 qa 录制到 a）
q         " 停止录制
@{寄存器}  " 播放宏（如 @a 播放 a 中记录的宏）
@@        " 重放上一次的宏
10@a      " 播放 a 宏 10 次

" 宏实战：在 10 行末尾加分号
" 1. 在第一行按 QA（开始录制宏到 a）
" 2. 按 $ 到行尾，按 a; 加入分号，按 Esc 回到普通模式
" 3. 按 j 到下一行，按 q（停止录制）
" 4. 按 10@a（在剩下的 10 行上播放宏）
```

### 17. 缩进与格式化

```vim
>>       " 向右缩进当前行
<<       " 向左缩进当前行
3>>      " 向右缩进 3 行
==       " 自动缩进当前行（根据语法？Vim 会帮你对齐！）
gg=G     " 自动缩进整个文件（格式化代码神器）

" 可视模式下
>        " 向右缩进选中区域
<        " 向左缩进选中区域
=        " 自动缩进选中区域
```

### 18. 组合命令实战

```vim
" 动作 + 动作对象 = 威力无穷
" 格式：{操作}{数字}{动作}

d2w      " 删除 2 个单词
3yy      " 复制 3 行
y2j      " 复制当前行及下面 2 行（共 3 行）
d/hello  " 从光标删到下一个 "hello"
c$       " 从光标改到行尾（删除并进入插入模式）
ci"      " 删除双引号内的内容（change inside quotes）—— 超实用！
ci(      " 删除括号内的内容
ci[      " 删除方括号内的内容
cit      " 删除 HTML 标签内的内容（change inside tag）
yi"      " 复制双引号内的内容
da"      " 删除包括双引号及其内部内容
diw      " 删除光标所在的整个单词（delete inner word）
daw      " 删除光标所在的单词及后面的空格
```

### 19. 文本对象（Text Objects）

```vim
" 配合编辑器使用：d=删除 c=修改 y=复制 v=选中

" 单词
iw       " 光标所在的单词（inner word）
aw       " 包含空格的单词（a word）

" 引号
i"       " 双引号内部
i'       " 单引号内部
i`       " 反引号内部

" 括号
i( 或 ib   " 小括号内部
i[         " 方括号内部
i{ 或 iB   " 花括号内部
i<         " 尖括号内部

" HTML/XML
it       " 标签内部（inner tag）
at       " 包含标签的整个区域

" 句子段落
is       " 当前句子
ip       " 当前段落
```

## 十、Vim 配置入门

```vim
" ～/.vimrc 常用配置（Windows 下是 ~/_vimrc 或 %USERPROFILE%/_vimrc）

" 基础设置
set nocompatible         " 关闭兼容模式（让 Vim 发挥全部能力）
syntax on               " 开启语法高亮
set number              " 显示行号
set relativenumber      " 显示相对行号（配合 d5j 这种操作）
set cursorline          " 高亮当前行
set showmatch           " 输入括号时高亮匹配的括号

" 缩进设置
set tabstop=4           " Tab 宽度 4 格
set shiftwidth=4        " 缩进宽度 4 格
set expandtab           " 用空格替代 Tab
set autoindent          " 自动缩进
set smartindent         " 智能缩进

" 搜索设置
set hlsearch            " 搜索高亮
set incsearch           " 边输入边搜索
set ignorecase          " 搜索忽略大小写
set smartcase           " 如果搜索词包含大写则区分大小写

" 体验优化
set mouse=a             " 允许鼠标操作
set clipboard=unnamed   " 使用系统剪贴板
set wildmenu            " 命令补全菜单
set wildmode=list:longest,full
set laststatus=2        " 始终显示状态栏
set background=dark     " 深色主题适配

" 编码设置（解决中文乱码）
set encoding=utf-8
set fileencodings=utf-8,gbk,gb2312,gb18030,cp936

" 设置 leader 键（默认是 \，很多人改成空格）
let mapleader = "\<Space>"
```
